cmake_minimum_required(VERSION 3.16)
project(TS3GhostUser VERSION 0.1.4 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(TS3_SDK_DIR "" CACHE PATH "Official ts3client-pluginsdk checkout")
if(NOT EXISTS "${TS3_SDK_DIR}/include/ts3_functions.h")
  message(FATAL_ERROR "Set TS3_SDK_DIR to a checkout of teamspeak/ts3client-pluginsdk (API 26)")
endif()
add_library(ghost_core STATIC src/controller.cpp src/udp_gate.cpp src/endpoint.cpp src/gui_tabs.cpp)
target_compile_features(ghost_core PUBLIC cxx_std_17)
target_include_directories(ghost_core PUBLIC src)
set_target_properties(ghost_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(ghostuser SHARED src/plugin.cpp)
target_compile_features(ghostuser PRIVATE cxx_std_17)
target_include_directories(ghostuser PRIVATE "${TS3_SDK_DIR}/include")
target_link_libraries(ghostuser PRIVATE ghost_core)
set_target_properties(ghostuser PROPERTIES PREFIX "")
if(WIN32)
  target_compile_definitions(ghost_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0601)
  target_link_libraries(ghost_core PUBLIC ws2_32)
  target_link_libraries(ghostuser PRIVATE comctl32 user32 gdi32)
  if(MSVC)
    set_property(TARGET ghost_core ghostuser PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(ghost_core PRIVATE /utf-8 /W4)
    target_compile_options(ghostuser PRIVATE /utf-8 /W4)
  elseif(MINGW)
    target_link_options(ghostuser PRIVATE -static-libgcc -static-libstdc++)
  endif()
else()
  target_compile_options(ghostuser PRIVATE -fvisibility=hidden -Wall -Wextra)
  target_compile_options(ghost_core PRIVATE -Wall -Wextra)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(plugin_menu_smoke tests/plugin_menu_smoke.cpp)
  target_compile_features(plugin_menu_smoke PRIVATE cxx_std_17)
  target_include_directories(plugin_menu_smoke PRIVATE src "${TS3_SDK_DIR}/include")
  if(WIN32)
    target_compile_definitions(plugin_menu_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(plugin_menu_smoke PRIVATE user32)
  else()
    target_link_libraries(plugin_menu_smoke PRIVATE ${CMAKE_DL_LIBS})
  endif()
  add_dependencies(plugin_menu_smoke ghostuser)
  add_test(NAME plugin_menu_smoke COMMAND plugin_menu_smoke "$<TARGET_FILE:ghostuser>" "${PROJECT_VERSION}")
  set_tests_properties(plugin_menu_smoke PROPERTIES TIMEOUT 30)
  add_executable(controller_test tests/controller_test.cpp)
  target_link_libraries(controller_test PRIVATE ghost_core)
  add_test(NAME controller_test COMMAND controller_test)
  add_executable(plugin_connection_test tests/plugin_connection_test.cpp)
  target_include_directories(plugin_connection_test PRIVATE "${TS3_SDK_DIR}/include")
  target_link_libraries(plugin_connection_test PRIVATE ghost_core)
  if(WIN32)
    target_link_libraries(plugin_connection_test PRIVATE comctl32 user32 gdi32)
    if(MSVC)
      target_compile_options(plugin_connection_test PRIVATE /utf-8 /W4)
    endif()
  endif()
  add_test(NAME plugin_connection_test COMMAND plugin_connection_test)
  if(WIN32 AND DEFINED ENV{TS3_QT_TEST_DIR})
    add_executable(qt_tabs_runtime_test tests/qt_tabs_runtime_test.cpp)
    target_compile_features(qt_tabs_runtime_test PRIVATE cxx_std_17)
    target_include_directories(qt_tabs_runtime_test PRIVATE src)
    target_compile_definitions(qt_tabs_runtime_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_link_libraries(qt_tabs_runtime_test PRIVATE user32)
    if(MSVC)
      target_compile_options(qt_tabs_runtime_test PRIVATE /utf-8 /W4)
    endif()
    add_test(NAME qt_tabs_runtime_test COMMAND qt_tabs_runtime_test "$ENV{TS3_QT_TEST_DIR}")
    set_tests_properties(qt_tabs_runtime_test PROPERTIES TIMEOUT 30
      ENVIRONMENT "QT_QPA_PLATFORM_PLUGIN_PATH=$ENV{TS3_QT_TEST_DIR}/platforms;QT_OPENGL=software;QT_DEBUG_PLUGINS=1")
  endif()
  if(NOT WIN32)
    add_executable(udp_gate_test tests/udp_gate_test.cpp)
    target_link_libraries(udp_gate_test PRIVATE ghost_core)
    add_test(NAME udp_gate_test COMMAND udp_gate_test)
  endif()
endif()
